# Runtime dependencies for the containerised inference service — nothing else.
#
# Install:
#   pip install --no-cache-dir -r requirements-prod.txt
#
# This file exists for two reasons that happen to coincide.
#
# 1. The running service imports a small fraction of requirements.txt. dvc,
#    deepchecks, locust, pytest, datasets and scikit-learn are development and
#    CI tools; shipping them enlarges the image, lengthens every pull during a
#    deployment, and widens the dependency surface of the only artifact that
#    faces the internet.
#
# 2. The default PyPI torch wheel bundles CUDA. Measured in a Python 3.12
#    virtual environment resolved from the unmodified requirements.txt:
#      nvidia-* libraries   2.7 GB   (15 separate packages)
#      triton                690 MB
#      torch itself          1.2 GB
#    Roughly 3.4 GB of GPU runtime installed for a service that runs on a
#    CPU-only Fargate task. The CPU-only build removes all of it.
#
# HOW THE CPU WHEEL IS SELECTED. The `+cpu` local version identifier is
# published only on PyTorch's own index, so pip cannot satisfy this requirement
# from PyPI and must take it from the extra index. That is deliberate — an
# unqualified `torch>=2.4.0` with the extra index present would still resolve
# to the CUDA build from PyPI, and the image would silently be 3 GB larger with
# nothing in the build log to say so.
#
# NOT VERIFIED HERE: download.pytorch.org is not reachable from the environment
# these files were written in, so the exact wheel's availability for
# cp312/linux was not confirmed by resolution. `make verify` (stage
# `image-size`) performs that resolution and records the before/after image
# sizes on the machine that builds the image.
--extra-index-url https://download.pytorch.org/whl/cpu

# Model runtime.
#   transformers is held below 5 for the same reason as in requirements.txt:
#   MLflow 2.16.2's transformers flavour imports FlaxPreTrainedModel, which
#   transformers 5 removed. app/utils.py calls mlflow.transformers.load_model
#   on the registry path, so the container needs the working combination too.
torch==2.4.1+cpu
transformers>=4.40.2,<5

# API.
fastapi==0.111.0
uvicorn[standard]==0.29.0
pydantic==2.7.1
python-dotenv==1.0.1

# Model registry client — required by app/utils.py when MODEL_SOURCE=mlflow.
# The full mlflow distribution rather than mlflow-skinny: skinny omits the
# transformers flavour that load_model needs.
mlflow==2.16.2

# Metrics exposition.
prometheus-client==0.20.0
